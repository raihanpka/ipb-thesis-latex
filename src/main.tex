%     main.tex                    Berkas ini (entry point)
%     config/                     Paket dan konfigurasi format
%       ipb-thesis.sty            Style utama (margins, font, headings, dll.)
%       information.tex           Data penulis, skripsi, institusi
%       hyphenation-id.tex        Aturan pemenggalan kata
%     abstract/                   Abstrak (id) dan Abstract (en)
%     preliminaries/              Bagian Awal (cover, pernyataan, dll.)
%     chapters/                   Bagian Inti (Bab I s.d. Bab V)
%     backmatter/                 Bagian Akhir (daftar pustaka, lampiran)
%     resources/                  Gambar, logo, dan sumber daya lain
%     refs/                       Database referensi BibTeX

% -----------------------------------------------------------------------------
% 1. Kelas Dokumen & Paket Utama
% -----------------------------------------------------------------------------
\documentclass[12pt,a4paper,twoside,openany]{book}

% Style utama (margins, font, headings, captions, dst.)
\usepackage{ipb-thesis}

% Data penulis, skripsi, institusi
\input{config/information}

% Aturan pemenggalan kata
\input{config/hyphenation-id}

% Database referensi untuk biblatex + biber
\addbibresource{refs/daftar-pustaka}

% -----------------------------------------------------------------------------
% 2. Page Styles
%    PPKI:
%      - Preliminary/backmatter: nomor halaman Romawi kecil, tengah atas
%      - Bab (mainmatter):       nomor halaman Arab, sudut luar atas
% -----------------------------------------------------------------------------
\fancypagestyle{frontmatter}{%
  \fancyhf{}%
  \fancyhead[C]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

\fancypagestyle{mainmatter}{%
  \fancyhf{}%
  \fancyhead[RO,LE]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyhead[RO,LE]{\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}%
}

% -----------------------------------------------------------------------------
% 3. Metadata untuk hyperref
% -----------------------------------------------------------------------------
\title{\JudulSkripsi}
\author{\NamaPenulis}
\date{\Tahun}

% -----------------------------------------------------------------------------
% 4. Titik Mulai Dokumen
% -----------------------------------------------------------------------------
\begin{document}

\noindent\ignorespaces

% -----------------------------------------------------------------------------
% 5. BAGIAN AWAL (PRELIMINARI) -- Bagian Awal
%    - Nomor halaman: Romawi kecil (i, ii, iii, ...), tengah atas
%    - Margin: 3 cm semua sisi + 1 cm gutter
% -----------------------------------------------------------------------------
\pagenumbering{roman}
\setcounter{page}{1}
\pagestyle{frontmatter}

\input{preliminaries/cover}
\input{preliminaries/pernyataan}
\input{preliminaries/lembar-persetujuan}
\input{abstract/abstract-id}
\input{abstract/abstract-en}
\input{preliminaries/prakata}
\input{preliminaries/daftar-isi}

% -----------------------------------------------------------------------------
% 6. BAGIAN INTI (BAB) -- Bagian Inti / Pokok
%    - Nomor halaman: Arab (1, 2, 3, ...), sudut luar atas
%    - Margin: 3-3-3-4 (4 cm sisi dalam untuk penjilidan)
% -----------------------------------------------------------------------------
\mainmatter
\setchaptermargins
\pagestyle{mainmatter}

\input{chapters/chapter-1}    % BAB I   PENDAHULUAN
\input{chapters/chapter-2}    % BAB II  TINJAUAN PUSTAKA
\input{chapters/chapter-3}    % BAB III METODE
\input{chapters/chapter-4}    % BAB IV  HASIL DAN PEMBAHASAN
\input{chapters/chapter-5}    % BAB V   SIMPULAN DAN SARAN

% -----------------------------------------------------------------------------
% 7. BAGIAN AKHIR -- Lampiran
%    - Nomor halaman: Arab (lanjut), sudut luar atas
%    - Margin kembali: 3 cm semua sisi + 1 cm gutter
% -----------------------------------------------------------------------------
\backmatter
\setpreliminarymargins
\pagestyle{mainmatter}

\input{backmatter/daftar-pustaka}
\input{backmatter/lampiran}
\input{backmatter/riwayat-hidup}

\end{document}
